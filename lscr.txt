-- ================= HÀM XỬ LÝ CHẠY SCRIPT ================= --
local function LoadExternalScript(scriptName, scriptUrl)
    Window:Notify({ 
        title = "Hệ thống", 
        content = "Đang tải: " .. tostring(scriptName), 
        duration = 2,
        icon = "rbxassetid://125823673784681"
    })

    task.spawn(function()
        local fetchSuccess, fetchResult = pcall(function() return game:HttpGet(scriptUrl) end)
        
        if not fetchSuccess then
            Window:Notify({ title = "Lỗi Mạng", content = "Không thể tải link script này!", duration = 4, icon = "rbxassetid://77861386382812" })
            return
        end
        
        local func, compileError = loadstring(fetchResult)
        if not func then
            Window:Notify({ title = "Lỗi Cú Pháp", content = "Script này đang bị lỗi code, không thể chạy!", duration = 4, icon = "rbxassetid://77861386382812" })
            return
        end
        
        local execSuccess, execError = pcall(func)
        if execSuccess then
            Window:Notify({ title = "Thành công", content = "Đã kích hoạt: " .. tostring(scriptName), duration = 3, icon = "rbxassetid://5578470911" })
        else
            Window:Notify({ title = "Lỗi Khi Chạy", content = "Script chạy bị lỗi! Hãy xem F9.", duration = 5, icon = "rbxassetid://77861386382812" })
        end
    end)
end

-- (Hệ thống lưu script cá nhân cục bộ)
local SaveFileName = "HzHub_SavedScripts.json"
local SavedScripts = {}

if isfile and isfile(SaveFileName) then
    local readSuccess, result = pcall(function() return HttpService:JSONDecode(readfile(SaveFileName)) end)

MusicTab:CreateSection({ name = "Music Player" })

local PlaybackToken = 0 
local function PlayMusic(id, songName)
    PlaybackToken = PlaybackToken + 1
    local currentToken = PlaybackToken
    CurrentSound:Stop()
    
    local cleanId = string.match(tostring(id), "%d+")
    if not cleanId then return end
    CurrentSound.SoundId = "rbxassetid://" .. cleanId

    task.spawn(function()
        local elapsed = 0
        while not CurrentSound.IsLoaded and elapsed < 1.5 do
            task.wait(0.1); elapsed = elapsed + 0.1
        end
        if currentToken == PlaybackToken and CurrentSound.IsLoaded then
            CurrentSound:Play()
            Window:Notify({ title = "Music", content = "Đang phát: " .. tostring(songName), duration = 3 })
        end
    end)
end

MusicTab:CreateDropdown({
    name = "Danh sách nhạc",
    options = MusicNames,
    value = {"None"},
    callback = function(selected)
        local choice = type(selected) == "table" and selected[1] or selected
        if choice == "None" then CurrentSound:Stop()
        elseif MusicMap[choice] then PlayMusic(MusicMap[choice], choice) end
    end
})

local OriginalMaterials = {}
local OriginalLighting = { GlobalShadows = Lighting.GlobalShadows, FogEnd = Lighting.FogEnd }

MiscTab:CreateToggle({
    name = "FPS Boost (Xoá Texture Client)",
    value = false,
    flag = "Settings_FPSBoost",
    callback = function(value)
        if value then
            Lighting.GlobalShadows = false
            Lighting.FogEnd = 9e9
            for _, v in pairs(workspace:GetDescendants()) do
                if v:IsA("BasePart") and not OriginalMaterials[v] then
                    OriginalMaterials[v] = { Material = v.Material, Reflectance = v.Reflectance }
                    v.Material = Enum.Material.SmoothPlastic
                    v.Reflectance = 0
                elseif v:IsA("Decal") or v:IsA("Texture") then
                    v.Transparency = 1
                end
            end
        else
            Lighting.GlobalShadows = OriginalLighting.GlobalShadows
            Lighting.FogEnd = OriginalLighting.FogEnd
            for part, data in pairs(OriginalMaterials) do
                if part and part.Parent then
                    part.Material = data.Material
                    part.Reflectance = data.Reflectance
                end
            end
            table.clear(OriginalMaterials)
            for _, v in pairs(workspace:GetDescendants()) do
                if (v:IsA("Decal") or v:IsA("Texture")) and v.Transparency == 1 then
                    v.Transparency = 0
                end
            end
        end
    end,
})
